========================================
101. СОЗДАНИЕ И
ИСПОЛЬЗОВАНИЕ PROMISE
========================================

Promise

—

объект,

который хранит

результат

асинхронной операции.

========================================
СОЗДАНИЕ
========================================

const myPromise =

new Promise(

(resolve, reject) => {

}

);

========================================
resolve()
========================================

Вызывается,

если операция

завершилась

успешно.

========================================
reject()
========================================

Вызывается,

если произошла

ошибка.

========================================
ПОЛУЧЕНИЕ
РЕЗУЛЬТАТА
========================================

myPromise

.then(...)

.catch(...)

========================================
.then()
========================================

Вызывается

после

resolve().

========================================
.catch()
========================================

Вызывается

после

reject().

========================================
СХЕМА
========================================

Promise

↓

Асинхронная
операция

↓

resolve()

или

reject()

↓

.then()

или

.catch()

========================================
ГЛАВНЫЕ ИДЕИ
========================================

Promise

хранит результат

будущей операции.

----------------------------------------

resolve()

—

успешное выполнение.

----------------------------------------

reject()

—

ошибка.

----------------------------------------

.then()

обрабатывает успех.

----------------------------------------

.catch()

обрабатывает ошибку.

========================================
