<!-- Пример HTML-файла -->
<!doctype html> <!-- Объявление типа документа (HTML5) -->
<html lang="ru"> <!-- Корневой элемент страницы, язык документа -->
  <head> <!-- Служебная информация о странице -->
    <title>Мой сайт</title> <!-- Заголовок страницы (вкладка браузера) -->
  </head>
  <body> <!-- Видимое содержимое страницы -->
    <h1>Привет!</h1> <!-- Заголовок первого уровня -->
  </body>
</html>
