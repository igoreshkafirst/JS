<!-- Атрибуты тегов HTML -->

src         <!-- Путь к файлу (img, script, audio, video и др.) -->
href        <!-- Адрес ссылки или подключаемого файла (a, link) -->
alt         <!-- Альтернативный текст для изображения -->
title       <!-- Всплывающая подсказка -->
class       <!-- Имя CSS-класса -->
id          <!-- Уникальный идентификатор элемента -->
style       <!-- Встроенные CSS-стили -->
lang        <!-- Язык документа или элемента -->
charset     <!-- Кодировка документа -->
name        <!-- Имя элемента формы или метатега -->
content     <!-- Содержимое метатега -->
type        <!-- Тип элемента (input, button, script и др.) -->
value       <!-- Значение элемента формы -->
placeholder <!-- Подсказка в поле ввода -->
disabled    <!-- Отключает элемент -->
checked     <!-- Отмечает checkbox или radio -->
selected    <!-- Выбирает пункт списка -->
required    <!-- Делает поле обязательным -->
readonly    <!-- Поле только для чтения -->
width       <!-- Ширина элемента -->
height      <!-- Высота элемента -->
target      <!-- Где открыть ссылку (_self, _blank и др.) -->
rel         <!-- Отношение между документами (link, a) -->

<!-- Атрибуты записываются внутри открывающего тега.
     Большинство атрибутов имеют вид:
     имя="значение"

     Некоторые атрибуты являются логическими
     и могут использоваться без значения:
     disabled, checked, selected, required -->
     
<!-- Пример использования атрибутов -->

<img
  src="img.jpg"
  class="rounded-img"
  alt="My Photo"
  title="Моя фотография"
/>

<!-- Минимальная структура HTML-документа -->

<!doctype html>
<html lang="en">
  <head>
    <meta charset="UTF-8" /> <!-- Кодировка UTF-8 -->
    <meta
      name="viewport"
      content="width=device-width, initial-scale=1.0"
    /> <!-- Адаптация под мобильные устройства -->
    <title>Document</title> <!-- Заголовок вкладки -->
  </head>

  <body>
    <!-- Содержимое страницы -->
  </body>
</html>
